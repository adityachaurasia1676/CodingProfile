class Solution {
    public boolean findPair(int[] arr, int x) {
        Arrays.sort(arr) ;
        for(int i=0;i<arr.length;i++){
            int val = arr[i] + x ;
            int l = 0 ;
            int h = arr.length - 1 ;
            while(l <= h){
                int mid = l + (h - l) / 2 ;
                
                if(arr[mid] == val && mid != i) return true ;
                
                if(arr[mid] > val){
                    h = mid - 1;
                }else{
                    l = mid + 1 ;
                }
            }
        }
        return false ;
    }
}
