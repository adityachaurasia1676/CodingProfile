class Solution {
    public int singleNumber(int[] nums) {
        Arrays.sort(nums);
        int[] arr = new int[nums.length - 1];
        for(int i=0;i<nums.length - 1;i++){
           if(nums[i] != nums[i + 1]){
                for(int j=0;j<nums.length;j++){
                    arr[j] = nums[i];
                }
           }
           
        } 
        return arr ;
    }
}
